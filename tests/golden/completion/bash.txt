# agit bash completion
_agit_complete() {
    local cur prev cmd
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"
    cmd="${COMP_WORDS[1]}"

    if [[ $COMP_CWORD -eq 1 ]]; then
        COMPREPLY=($(compgen -W "-h --help -V --version init observe uninstall doctor fsck gc push pull export import status timeline sessions log restore show diff between grep blame cat privacy reindex version completion" -- "$cur"))
        return 0
    fi

    case "$cmd" in
    init)
        if [[ "$prev" == "--agent" ]]; then
            COMPREPLY=($(compgen -W "claude codex gemini" -- "$cur"))
            return 0
        fi
        COMPREPLY=($(compgen -W "--agent --dry-run --force -h --help " -- "$cur"))
        ;;
    observe)
        COMPREPLY=($(compgen -W "--json --once --input -h --help " -- "$cur"))
        ;;
    uninstall)
        COMPREPLY=($(compgen -W "-h --help " -- "$cur"))
        ;;
    doctor)
        COMPREPLY=($(compgen -W "--json --locks --stats --last-hook-error -h --help " -- "$cur"))
        ;;
    fsck)
        COMPREPLY=($(compgen -W "--json --reindex -h --help " -- "$cur"))
        ;;
    gc)
        COMPREPLY=($(compgen -W "--json --grace-hours --prune-before -h --help " -- "$cur"))
        ;;
    push)
        COMPREPLY=($(compgen -W "--json --remote --allow-sensitive -h --help " -- "$cur"))
        ;;
    pull)
        COMPREPLY=($(compgen -W "--json --remote -h --help " -- "$cur"))
        ;;
    export)
        COMPREPLY=($(compgen -W "--json --origin --session --since --until --allow-sensitive -h --help " -- "$cur"))
        ;;
    import)
        COMPREPLY=($(compgen -W "--json --replace-ref -h --help " -- "$cur"))
        ;;
    status)
        COMPREPLY=($(compgen -W "--json -h --help " -- "$cur"))
        ;;
    timeline)
        COMPREPLY=($(compgen -W "--origin --session --since --until --limit --redacted --full -h --help " -- "$cur"))
        ;;
    sessions)
        COMPREPLY=($(compgen -W "--json -h --help " -- "$cur"))
        ;;
    log)
        COMPREPLY=($(compgen -W "--json -h --help " -- "$cur"))
        ;;
    restore)
        COMPREPLY=($(compgen -W "--json --all --force --dry-run -h --help " -- "$cur"))
        ;;
    show)
        COMPREPLY=($(compgen -W "--json --files --stat --redacted --full -h --help " -- "$cur"))
        ;;
    diff)
        COMPREPLY=($(compgen -W "--redacted --full -h --help " -- "$cur"))
        ;;
    between)
        COMPREPLY=($(compgen -W "--json -h --help " -- "$cur"))
        ;;
    grep)
        COMPREPLY=($(compgen -W "--json --origin --session --since --until --limit --context --redacted --full -h --help " -- "$cur"))
        ;;
    blame)
        COMPREPLY=($(compgen -W "--no-limits -h --help " -- "$cur"))
        ;;
    cat)
        COMPREPLY=($(compgen -W "--redacted --full -h --help " -- "$cur"))
        ;;
    privacy)
        COMPREPLY=($(compgen -W "--json -h --help " -- "$cur"))
        ;;
    reindex)
        COMPREPLY=($(compgen -W "--from -h --help " -- "$cur"))
        ;;
    version)
        COMPREPLY=($(compgen -W "-h --help " -- "$cur"))
        ;;
    completion)
        COMPREPLY=($(compgen -W "-h --help " -- "$cur"))
        ;;
    esac
}
complete -F _agit_complete agit
