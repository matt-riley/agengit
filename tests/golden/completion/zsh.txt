#compdef agit
_agit() {
    local context state line
    local -a commands
    commands=(
        'init:Set up agit in the current repository'
        'uninstall:Remove agit hooks from agent configurations'
        'doctor:Check agent hook configurations and store health'
        'fsck:Verify store integrity without mutating data'
        'gc:Prune unreachable store data and stale temporaries'
        'push:Upload store history to a configured remote'
        'pull:Download store history from a configured remote'
        'export:Write a portable bundle from selected sessions'
        'import:Import a portable bundle into the local store'
        'status:Show the investigation dashboard'
        'timeline:Show recent recorded steps'
        'sessions:List recorded agent sessions'
        'log:Show step history for a session'
        'show:Show details of a step'
        'diff:Show a text diff for a step'
        'grep:Search recorded messages and tool activity'
        'blame:Show per-line step attribution for a file'
        'cat:Print a raw object by hash'
        'privacy:Scan reachable content for sensitive data'
        'reindex:Rebuild the index from the object store'
        'version:Print version information'
        'completion:Generate shell completion scripts'
    )
    _arguments -C \
        '-h[Display this help and exit.]' \
        '-V[Print version and exit.]' \
        '1:command:->command' \
        '*::arg:->args'

    case $state in
        command)
            _describe 'agit command' commands
            ;;
        args)
            case $words[2] in
                init)
                    _arguments \
                        '--agent[Install only the specified agent (claude, codex, gemini). Can be repeated.]:name:(claude codex gemini)' \
                        '--dry-run[Show what would be installed without making changes.]' \
                        '--force[Back up and replace malformed/non-object existing JSON config.]' \
                        '--help[Display this help and exit.]'
                    ;;
                uninstall)
                    _arguments \
                        '--help[Display this help and exit.]'
                    ;;
                doctor)
                    _arguments \
                        '--json[Render doctor checks as structured JSON.]' \
                        '--locks[List currently held lock files with age and executable path.]' \
                        '--stats[Print finalize retry/object-write counters.]' \
                        '--last-hook-error[Pretty-print the newest hook-error log entry.]' \
                        '--help[Display this help and exit.]'
                    ;;
                fsck)
                    _arguments \
                        '--json[Render fsck checks as structured JSON.]' \
                        '--reindex[Explicitly rebuild only index.db after a successful read-only scan.]' \
                        '--help[Display this help and exit.]'
                    ;;
                gc)
                    _arguments \
                        '--json[Render gc results as structured JSON.]' \
                        '--grace-hours[Only prune unreachable objects/tmp files older than <N> hours. Defaults to 2.]:N' \
                        '--prune-before[Delete session refs whose head-step timestamp is older than UTC midnight on the given date before sweeping.]:YYYY-MM-DD' \
                        '--help[Display this help and exit.]'
                    ;;
                push)
                    _arguments \
                        '--json[Render push results as structured JSON.]' \
                        '--remote[Choose a configured remote by name.]:name' \
                        '--allow-sensitive[Allow plaintext uploads even when privacy scan findings exist.]' \
                        '--help[Display this help and exit.]'
                    ;;
                pull)
                    _arguments \
                        '--json[Render pull results as structured JSON.]' \
                        '--remote[Choose a configured remote by name.]:name' \
                        '--help[Display this help and exit.]'
                    ;;
                export)
                    _arguments \
                        '--json[Render export results as structured JSON.]' \
                        '--origin[Only export sessions recorded by the given origin.]:name' \
                        '--session[Only export one disambiguated session id.]:origin/session-id' \
                        '--since[Only select sessions with at least one step on or after UTC midnight for the given date.]:YYYY-MM-DD' \
                        '--until[Only select sessions with at least one step before UTC midnight after the given date.]:YYYY-MM-DD' \
                        '--allow-sensitive[Allow plaintext export when privacy scan findings exist; writes a privacy report.]' \
                        '--help[Display this help and exit.]'
                    ;;
                import)
                    _arguments \
                        '--json[Render import results as structured JSON.]' \
                        '--replace-ref[Overwrite a named canonical session ref instead of namespacing conflicts. Can be repeated.]:origin/session-id' \
                        '--help[Display this help and exit.]'
                    ;;
                status)
                    _arguments \
                        '--json[Render the status as structured JSON.]' \
                        '--help[Display this help and exit.]'
                    ;;
                timeline)
                    _arguments \
                        '--origin[Only show steps recorded by the given origin.]:name' \
                        '--session[Only show one session id, or pass origin/session-id to disambiguate.]:id' \
                        '--since[Only include steps on or after UTC midnight for the given date.]:YYYY-MM-DD' \
                        '--until[Only include steps before UTC midnight after the given date.]:YYYY-MM-DD' \
                        '--limit[Show at most <N> steps. Defaults to 20.]:N' \
                        '--redacted[Redact obvious secrets in previews.]' \
                        '--full[Render full previews even when redaction is the repo default.]' \
                        '--help[Display this help and exit.]'
                    ;;
                sessions)
                    _arguments \
                        '--json[Render the session list as structured JSON.]' \
                        '--help[Display this help and exit.]'
                    ;;
                log)
                    _arguments \
                        '--json[Render the session log as structured JSON.]' \
                        '--help[Display this help and exit.]'
                    ;;
                show)
                    _arguments \
                        '--json[Render the step details as structured JSON.]' \
                        '--files[List captured files from the step tree.]' \
                        '--stat[Summarize file changes between this step and its parent.]' \
                        '--redacted[Redact obvious secrets in the rendered output.]' \
                        '--full[Render full output even when redaction is the repo default.]' \
                        '--help[Display this help and exit.]'
                    ;;
                diff)
                    _arguments \
                        '--redacted[Redact obvious secrets in diff output.]' \
                        '--full[Render full diff output even when redaction is the repo default.]' \
                        '--help[Display this help and exit.]'
                    ;;
                grep)
                    _arguments \
                        '--json[Render grep matches as structured JSON.]' \
                        '--origin[Only search matches recorded by the given origin.]:name' \
                        '--session[Only search one session id, or pass origin/session-id to disambiguate.]:id' \
                        '--since[Only include matches on or after UTC midnight for the given date.]:YYYY-MM-DD' \
                        '--until[Only include matches before UTC midnight after the given date.]:YYYY-MM-DD' \
                        '--limit[Return at most <N> matches. Defaults to 20.]:N' \
                        '--context[Use <N> snippet tokens around each match. Defaults to 12.]:N' \
                        '--redacted[Redact obvious secrets in rendered snippets.]' \
                        '--full[Render full snippets even when redaction is the repo default.]' \
                        '--help[Display this help and exit.]'
                    ;;
                blame)
                    _arguments \
                        '--no-limits[Disable the blame file-size cap for this invocation when blame output is available.]' \
                        '--help[Display this help and exit.]'
                    ;;
                cat)
                    _arguments \
                        '--redacted[Redact obvious secrets in the rendered object output.]' \
                        '--full[Render full object output even when redaction is the repo default.]' \
                        '--help[Display this help and exit.]'
                    ;;
                privacy)
                    _arguments \
                        '--json[Render privacy scan findings as structured JSON.]' \
                        '--help[Display this help and exit.]'
                    ;;
                reindex)
                    _arguments \
                        '--from[Incrementally replay steps newer than <HASH> that are reachable from session refs.]:HASH' \
                        '--help[Display this help and exit.]'
                    ;;
                version)
                    _arguments \
                        '--help[Display this help and exit.]'
                    ;;
                completion)
                    _arguments \
                        '--help[Display this help and exit.]'
                    ;;
            esac
            ;;
    esac
}
_agit "$@"
