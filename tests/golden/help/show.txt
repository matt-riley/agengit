USAGE:
    agit show [OPTIONS] <HASH>

DESCRIPTION:
    Show details of a recorded step object by its BLAKE3 hash.

OPTIONS:
    --json
        Render the step details as structured JSON.
    --files
        List captured files from the step tree.
    --stat
        Summarize file changes between this step and its parent.
    --redacted
        Redact obvious secrets in the rendered output.
    --full
        Render full output even when redaction is the repo default.
    -h, --help
        Display this help and exit.

EXAMPLES:
    # show details of a step
    agit show abc123def
    # show captured files for a step
    agit show --files abc123def
    # show file-change counts for a step
    agit show --stat abc123def
